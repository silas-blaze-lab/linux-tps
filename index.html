TP 4 : Programmation Shell (Scripts)

Script 1 : Calcul de moyenne et mentions 
#!/bin/bash
sum=0; count=0
while true; do
    read -p "Saisir une note (ou 'q' pour quitter) : " input
    [[ "$input" == "q" ]] && break
    if (( $(echo "$input < 10" | bc -l) )); then echo "Ajourné"
    elif (( $(echo "$input < 12" | bc -l) )); then echo "Passable"
    else echo "Bien"; fi
    sum=$(echo "$sum + $input" | bc -l); count=$((count + 1))
done
[ $count -gt 0 ] && echo "Moyenne: $(echo "$sum/$count" | bc -l)"
Script 2 : Vérification d'existence d'utilisateur 
#!/bin/bash
read -p "Login : " user
grep -q "^$user:" /etc/passwd && echo "Existe" || echo "N'existe pas"
Script 3 : Affichage des membres d'un groupe 
#!/bin/bash
read -p "Groupe : " grp
grep "^$grp:" /etc/group | cut -d: -f4
Script 4 : Copie sécurisée avec paramètres 
#!/bin/bash
if [ $# -ne 2 ]; then echo "Usage: $0 source destination"; exit 1; fi
[ -f "$1" ] && cp "$1" "$2" && echo "Copié" || echo "Erreur"
Script 5 : Modification interactive des droits 
#!/bin/bash
read -p "Fichier : " f; read -p "Droits (ex: 755) : " d
[ -e "$f" ] && chmod "$d" "$f" && ls -l "$f"
